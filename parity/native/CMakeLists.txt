  message(FATAL_ERROR "Set TAGLIB_SOURCE to the upstream source directory")

file(GLOB_RECURSE headers "${TAGLIB_SOURCE}/taglib/*.h")
set(include_dirs "${TAGLIB_SOURCE}/taglib" "${TAGLIB_SOURCE}/taglib/toolkit" "${CMAKE_CURRENT_BINARY_DIR}/upstream" "${CMAKE_CURRENT_SOURCE_DIR}/core")
foreach(header IN LISTS headers)
  get_filename_component(directory "${header}" DIRECTORY)
  list(APPEND include_dirs "${directory}")
endforeach()
list(REMOVE_DUPLICATES include_dirs)
# Keep the compilation settings of the previous independent consumers.
function(probe name source)
  add_executable(${name} ${source})
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
  target_include_directories(${name} PRIVATE ${include_dirs})
  target_compile_definitions(${name} PRIVATE TAGLIB_STATIC)
  target_compile_options(${name} PRIVATE -O0 -UNDEBUG)
  target_link_libraries(${name} PRIVATE tag Threads::Threads)
endfunction()
